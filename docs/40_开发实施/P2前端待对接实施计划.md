```typescript
import FileUploader from '@/components/common/FileUploader.vue'
import { uploadFeedbackAttachment, deleteFeedbackAttachment } from '@/api/feedback'
import { ElMessageBox } from 'element-plus'

const attachmentUploading = ref(false)

/** FileUploader v-model 变更：新增的 url 逐个转存为反馈附件 */
async function handleAttachmentsChange(urls: string[]): Promise<void> {
  const existing = new Set(attachments.value.map((item) => item.fileUrl))
  const added = urls.filter((url) => !existing.has(url))
  if (added.length === 0) return
  attachmentUploading.value = true
  let failed = 0
  for (const url of added) {
    try {
      const blob = await (await fetch(url)).blob()
      const fileName = url.split('/').pop() ?? 'attachment'
      await uploadFeedbackAttachment(feedbackId, new File([blob], fileName, { type: blob.type }))
    } catch {
      failed += 1
    }
  }
  attachmentUploading.value = false
  if (failed > 0) ElMessage.warning(`${failed} 个附件关联失败，请重试`)
  await loadAttachments()
}

async function handleDeleteAttachment(id: number): Promise<void> {
  try {
    await ElMessageBox.confirm('确认删除该附件？', '提示', { type: 'warning' })
  } catch {
    return
  }
  try {
    await deleteFeedbackAttachment(id)
    ElMessage.success('附件已删除')
    await loadAttachments()
  } catch (error) {
    ElMessage.error(error instanceof Error ? error.message : '删除失败')
  }
}
```

```typescript
import SockJS from 'sockjs-client'
import { Client } from '@stomp/stompjs'
import { getToken } from './auth'

/**
 * WebSocket 通知连接管理（接口设计.md 9.11.1.6）：
 * SockJS + STOMP，CONNECT 帧 JWT 认证，指数退避重连；
 * 推送失败/断线由 store/notification.ts 的轮询兜底
 */

type NotificationHandler = (payload: unknown) => void

let client: Client | null = null
let handler: NotificationHandler | null = null
let reconnectDelay = 2000

/** 连接并订阅用户通知队列（重复调用幂等） */
export function connectWebSocket(onNotification: NotificationHandler): void {
  handler = onNotification
  if (client?.active) return

  const token = getToken()
  if (!token) return

  client = new Client({
    webSocketFactory: () => new SockJS('/ws') as WebSocket,
    connectHeaders: { Authorization: `Bearer ${token}` },
    reconnectDelay: 0,
    onConnect: () => {
      reconnectDelay = 2000
      client?.subscribe('/user/queue/notifications', (frame) => {
        try {
          handler?.(JSON.parse(frame.body))
        } catch {
          /* 载荷解析失败丢弃，轮询兜底 */
        }
      })
    },
    onWebSocketClose: () => {
      scheduleReconnect()
    },
    onStompError: () => {
      /* 认证失败等：静默停止，由轮询兜底 */
      client?.deactivate()
      client = null
    }
  })
  client.activate()
}

function scheduleReconnect(): void {
  if (!client) return
  const delay = reconnectDelay
  reconnectDelay = Math.min(reconnectDelay * 2, 30000)
  setTimeout(() => {
    if (handler) connectWebSocket(handler)
  }, delay)
}

/** 断开连接（登出时调用） */
export function disconnectWebSocket(): void {
  handler = null
  client?.deactivate()
  client = null
  reconnectDelay = 2000
}
```

## 任务 5：FE-T3 通知 store + 登录登出集成

**文件：**
- 创建：`frontend/src/store/notification.ts`
- 修改：`frontend/src/store/user.ts`（setSession 连接 / logout 断开）

**背景：** 全新 store（现有 `store/` 只有 index/user/permission）。载荷
`isRead` 是 0/1 数字，统一在 addNotification 转布尔。轮询兜底：WS 断开时
每 30s 调 `pullNotifications(lastSeq)`。

- [x] **步骤 5.1：新建 store/notification.ts**

```typescript
import { computed, ref } from 'vue'
import { defineStore } from 'pinia'
import type { INotification, IPulledNotification } from '@/types/modules/notification'
import { pullNotifications, listUnreadNotifications } from '@/api/notification'
import { connectWebSocket, disconnectWebSocket } from '@/utils/websocket'

/**
 * 通知中心状态：WS 实时推送 + 断线轮询兜底；
 * seq 全局递增，去重与补拉都以 seq 为游标
 */

const POLL_INTERVAL = 30000
const POLL_FALLBACK_ONLY = 10000

export const useNotificationStore = defineStore('notification', () => {
  const notifications = ref<INotification[]>([])
  const unreadCount = ref(0)
  const lastSeq = ref(0)
  let pollTimer: ReturnType<typeof setInterval> | null = null

  const latest = computed(() => notifications.value.slice(0, 20))

  /** WS 推送载荷（NotificationVO：isRead 为 0/1） */
  function handlePush(payload: IPulledNotification & { isRead: number }): void {
    if (notifications.value.some((n) => n.id === payload.id || n.seq === payload.seq)) return
    notifications.value.unshift({
      ...payload,
      isRead: payload.isRead === 1
    })
    if (payload.isRead !== 1) unreadCount.value += 1
    lastSeq.value = Math.max(lastSeq.value, Number(payload.seq))
  }

  function startPolling(interval: number): void {
    stopPolling()
    pollTimer = setInterval(async () => {
      try {
        const result = await pullNotifications(lastSeq.value)
        for (const item of result.notifications) {
          if (!notifications.value.some((n) => n.id === item.id)) {
            notifications.value.unshift(item)
            if (!item.isRead) unreadCount.value += 1
          }
        }
        lastSeq.value = Math.max(lastSeq.value, result.latestSeq)
        /* 未读数以服务端口径为准（WS 推送未读计数可能漂移） */
        const unread = await listUnreadNotifications()
        unreadCount.value = unread.count
      } catch {
        /* 轮询失败下轮重试 */
      }
    }, interval)
  }

  function stopPolling(): void {
    if (pollTimer !== null) {
      clearInterval(pollTimer)
      pollTimer = null
    }
  }

  /** 登录后初始化（user store setSession 调用） */
  function init(): void {
    connectWebSocket(handlePush)
    /* WS 在线时低频轮询兜底，断线后 STOMP 自身重连；轮询始终作为第二保障 */
    startPolling(POLL_INTERVAL)
  }

  /** 登出清理（user store logout 调用） */
  function cleanup(): void {
    disconnectWebSocket()
    stopPolling()
    notifications.value = []
    unreadCount.value = 0
    lastSeq.value = 0
  }

  /** 通知页/铃铛主动刷新未读数 */
  async function refreshUnread(): Promise<void> {
    try {
      const result = await listUnreadNotifications()
      unreadCount.value = result.count
    } catch {
      /* 忽略 */
    }
  }

  function markLocalRead(id: number): void {
    const item = notifications.value.find((n) => n.id === id)
    if (item && !item.isRead) {
      item.isRead = true
      unreadCount.value = Math.max(0, unreadCount.value - 1)
    }
  }

  function markAllLocalRead(): void {
    notifications.value.forEach((n) => (n.isRead = true))
    unreadCount.value = 0
  }

  return {
    notifications, latest, unreadCount, lastSeq,
    init, cleanup, refreshUnread, markLocalRead, markAllLocalRead, startPolling, stopPolling,
    POLL_FALLBACK_ONLY
  }
})
```
