**步骤 3：创建 store/user.ts（用户状态）**

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'

export interface UserInfo {
  userId: string
  username: string
  role: 'SUPER_ADMIN' | 'ADMIN' | 'STAFF' | 'RESIDENT' | 'GUEST'
  communityIds?: string[] // ADMIN 角色绑定的社区列表
}

export const useUserStore = defineStore('user', () => {
  const token = ref<string>('')
  const userInfo = ref<UserInfo | null>(null)
  const isLoggedIn = ref<boolean>(false)

  // 设置 Token
  const setToken = (newToken: string) => {
    token.value = newToken
    localStorage.setItem('token', newToken)
  }

  // 设置用户信息
  const setUserInfo = (info: UserInfo) => {
    userInfo.value = info
    isLoggedIn.value = true
  }

  // 清除登录状态
  const logout = () => {
    token.value = ''
    userInfo.value = null
    isLoggedIn.value = false
    localStorage.removeItem('token')
  }

  // 初始化（从 localStorage 恢复）
  const init = () => {
    const savedToken = localStorage.getItem('token')
    if (savedToken) {
      token.value = savedToken
      isLoggedIn.value = true
    }
  }

  return {
    token,
    userInfo,
    isLoggedIn,
    setToken,
    setUserInfo,
    logout,
    init,
  }
})
```

**步骤 4：创建 store/permission.ts（权限状态）**

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { useUserStore } from './user'
import type { RouteRecordRaw } from 'vue-router'

export const usePermissionStore = defineStore('permission', () => {
  const userStore = useUserStore()
  
  const permissionRoutes = ref<RouteRecordRaw[]>([])
  const buttonPermissions = ref<Set<string>>(new Set())

  // 检查是否有某个权限
  const hasPermission = (permission: string): boolean => {
    return buttonPermissions.value.has(permission)
  }

  // 检查是否有某个角色
  const hasRole = (role: string | string[]): boolean => {
    const userRole = userStore.userInfo?.role
    if (!userRole) return false
    
    if (Array.isArray(role)) {
      return role.includes(userRole)
    }
    return userRole === role
  }

  // 设置路由权限
  const setPermissionRoutes = (routes: RouteRecordRaw[]) => {
    permissionRoutes.value = routes
  }

  // 设置按钮权限
  const setButtonPermissions = (permissions: string[]) => {
    buttonPermissions.value = new Set(permissions)
  }

  return {
    permissionRoutes,
    buttonPermissions,
    hasPermission,
    hasRole,
    setPermissionRoutes,
    setButtonPermissions,
  }
})
```

```typescript
import SockJS from 'sockjs-client'
import { Client, IMessage } from '@stomp/stompjs'
import { useUserStore } from '@/store/user'

export interface NotificationMessage {
  id: string
  seq: number
  type: string
  title: string
  content: string
  createdAt: string
}

class WebSocketClient {
  private client: Client | null = null
  private connected = false
  private reconnectAttempts = 0
  private maxReconnectAttempts = 5
  private reconnectDelay = 3000

  // 消息回调
  private onMessageCallback: ((message: NotificationMessage) => void) | null = null
  private onConnectedCallback: (() => void) | null = null
  private onDisconnectedCallback: (() => void) | null = null

  connect() {
    const userStore = useUserStore()
    if (!userStore.token) {
      console.warn('WebSocket: 未登录，跳过连接')
      return
    }

    this.client = new Client({
      webSocketFactory: () => new SockJS('/ws'),
      connectHeaders: {
        Authorization: `Bearer ${userStore.token}`,
      },
      debug: (str) => {
        console.log('[WebSocket Debug]', str)
      },
      reconnectDelay: this.reconnectDelay,
      heartbeatIncoming: 4000,
      heartbeatOutgoing: 4000,
    })

    this.client.onConnect = () => {
      console.log('WebSocket 连接成功')
      this.connected = true
      this.reconnectAttempts = 0
      
      // 订阅个人通知队列
      this.client?.subscribe('/user/queue/notifications', (message: IMessage) => {
        const notification: NotificationMessage = JSON.parse(message.body)
        console.log('收到通知:', notification)
        this.onMessageCallback?.(notification)
      })

      this.onConnectedCallback?.()
    }

    this.client.onDisconnect = () => {
      console.log('WebSocket 断开连接')
      this.connected = false
      this.onDisconnectedCallback?.()
    }

    this.client.onStompError = (frame) => {
      console.error('WebSocket STOMP 错误:', frame.headers['message'])
      console.error('详细信息:', frame.body)
    }

    this.client.activate()
  }

  disconnect() {
    if (this.client) {
      this.client.deactivate()
      this.client = null
      this.connected = false
    }
  }

  // 注册消息回调
  onMessage(callback: (message: NotificationMessage) => void) {
    this.onMessageCallback = callback
  }

  onConnected(callback: () => void) {
    this.onConnectedCallback = callback
  }

  onDisconnected(callback: () => void) {
    this.onDisconnectedCallback = callback
  }

  isConnected(): boolean {
    return this.connected
  }
}

export const wsClient = new WebSocketClient()
```

```typescript
  }
)

// 响应拦截器
service.interceptors.response.use(
  (response: AxiosResponse) => {
    const { code, message, data } = response.data
    
    if (code === 200) {
      return data
    } else {
      ElMessage.error(message || '请求失败')
      return Promise.reject(new Error(message || '请求失败'))
    }
  },
  (error) => {
    if (error.response?.status === 401) {
      const userStore = useUserStore()
      userStore.logout()
      router.push('/auth/login')
      ElMessage.error('登录已过期，请重新登录')
    } else {
      ElMessage.error(error.response?.data?.message || '网络错误')
    }
    return Promise.reject(error)
  }
)

export default service
```

## 6. 问题排查

### 6.1 Element Plus 样式未生效

**现象**：组件显示但样式不正确

**排查**：
1. 检查是否引入样式文件（`import 'element-plus/dist/index.css'`）
2. 检查 Vite 配置是否正确处理 CSS
3. 清除浏览器缓存重试

### 6.2 Pinia 状态丢失

**现象**：刷新页面后状态丢失

**排查**：
1. 检查是否使用 localStorage 持久化
2. 检查 userStore.init() 是否在 main.ts 调用
3. 检查浏览器是否禁用 localStorage

### 6.3 环境变量未生效

**现象**：前端使用默认配置而非环境变量

**排查**：
1. 检查环境变量文件名是否正确（`.env.development`）
2. 检查环境变量前缀是否为 `VITE_`（Vite 要求）
3. 重启开发服务器（环境变量修改需要重启）
4. 使用 `import.meta.env.VITE_*` 访问环境变量

### 6.4 TypeScript 类型错误

**现象**：`npm run typecheck` 报错

**排查**：
1. 检查 types 目录下类型定义是否完整
2. 检查 tsconfig.json 配置是否正确
3. 检查是否有未定义的全局类型
4. 运行 `npm install` 重新安装依赖
