### 3.3 API 封装规范

**API 文件结构**（以 `workorder.ts` 为例）：

```typescript
import request from '@/utils/request'
import type { IWorkOrder, IWorkOrderDetail, ICreateWorkOrderDTO } from '@/types/api'

/** C4 工单管理接口 */
export default {
  /** 9.4.2.1 创建工单 */
  create(data: ICreateWorkOrderDTO) {
    return request.post<IWorkOrder>('/api/v1/work-orders', data)
  },

  /** 9.4.2.2 查询工单列表（分页） */
  getList(params: {
    page?: number
    size?: number
    status?: string
    categoryId?: number
    keyword?: string
  }) {
    return request.get<{ records: IWorkOrder[]; total: number }>('/api/v1/work-orders', { params })
  },

  /** 9.4.2.3 查询工单详情 */
  getDetail(id: number) {
    return request.get<IWorkOrderDetail>(`/api/v1/work-orders/${id}`)
  },

  /** 9.4.2.7 派单 */
  assign(id: number, data: { staffId: number; remark?: string }) {
    return request.patch<void>(`/api/v1/work-orders/${id}/assign`, data)
  },

  /** 9.4.2.8 接单 */
  accept(id: number) {
    return request.patch<void>(`/api/v1/work-orders/${id}/accept`)
  },

  /** 9.4.2.9 处理工单 */
  process(id: number, data: { result: string; images?: string[] }) {
    return request.patch<void>(`/api/v1/work-orders/${id}/process`, data)
  },

  /** 9.4.2.10 确认完成 */
  confirm(id: number, data: { satisfied: boolean; remark?: string }) {
    return request.patch<void>(`/api/v1/work-orders/${id}/confirm`, data)
  }
}
```

### 3.5 权限控制规范

**路由守卫**（`router/index.ts`）：

```typescript
router.beforeEach((to, from, next) => {
  const userStore = useUserStore()
  const token = userStore.token

  // 1. 公开路由（无需登录）
  const publicRoutes = ['/guest/home', '/guest/housings', '/guest/notices', '/auth/login', '/auth/register']
  if (publicRoutes.includes(to.path)) {
    return next()
  }

  // 2. 未登录跳转登录页
  if (!token) {
    return next('/auth/login')
  }

  // 3. 角色权限检查
  const requiredRoles = to.meta.roles as string[] | undefined
  if (requiredRoles && !requiredRoles.includes(userStore.role)) {
    return next('/403')
  }

  // 4. 角色首页重定向
  if (to.path === '/') {
    const homeMap: Record<string, string> = {
      RESIDENT: '/resident/home',
      STAFF: '/staff/dashboard',
      ADMIN: '/admin/dashboard',
      SUPER_ADMIN: '/admin/dashboard'
    }
    return next(homeMap[userStore.role] || '/auth/login')
  }

  next()
})
```

## 8. 风险与注意事项

### 8.1 技术风险

1. **Element Plus 样式覆盖**：
   - 风险：全局引入 Element Plus 样式可能与自定义样式冲突
   - 缓解：使用 CSS Module 或 BEM 命名规范，避免样式污染

2. **WebSocket 断线重连**：
   - 风险：WebSocket 连接不稳定，断线后需重连
   - 缓解：P1 阶段用 HTTP 轮询，P2 阶段再实现 WebSocket + 断线重连

3. **ECharts 图表性能**：
   - 风险：大数据量图表渲染性能差
   - 缓解：按需加载 ECharts 模块，使用数据抽样/懒加载

### 8.2 进度风险

1. **页面数量多**：
   - 风险：64 个页面开发周期长
   - 缓解：按优先级分批开发，优先完成核心流程

2. **前后端联调依赖**：
   - 风险：后端接口未完成，前端无法联调
   - 缓解：前端 Mock 数据开发，后端接口完成后再联调
