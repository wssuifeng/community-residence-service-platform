INSERT IGNORE INTO time_slot_template (resource_id, day_of_week, start_time, end_time, max_capacity, created_at, updated_at)
SELECT 1 AS resource_id, 
       d.day_of_week,
       '09:00:00' AS start_time,
       '21:00:00' AS end_time,
       1 AS max_capacity,
       NOW() AS created_at,
       NOW() AS updated_at
FROM (
  SELECT 1 AS day_of_week UNION ALL SELECT 2 UNION ALL SELECT 3 UNION ALL 
  SELECT 4 UNION ALL SELECT 5 UNION ALL SELECT 6 UNION ALL SELECT 7

  SELECT 1 FROM time_slot_template WHERE resource_id = 1
);
